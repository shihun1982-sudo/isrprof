# C++ modem sample (ARM cross build in the real project; here only the compile database matters)
cmake_minimum_required(VERSION 3.16)
project(cxx_modem CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
file(GLOB_RECURSE SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_library(cxx_modem STATIC ${SRC})
target_include_directories(cxx_modem PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(cxx_modem PUBLIC CONFIG_NR_MU=1)
